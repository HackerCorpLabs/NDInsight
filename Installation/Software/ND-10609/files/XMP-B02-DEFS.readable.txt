%-----------------------------------------------------------------------------%
%                                                                             %
%                    format of XROUT messages                                 %
%                                                                             %
%-----------------------------------------------------------------------------%

%
% maximum XMSG name length
%
CONSTANT XMMAXNameLength = 32          % actually installation-dependent!


                                       % XROUT header format
TYPE XROUTHDR = RECORD PACK            % base record for variants
       BYTE: XRSerialNumber            % serial number for user's use
       BYTE: XRServiceNumber           % code for required service which is
       BYTE: XRStatus = XRServiceNumber% replaced with XROUT status code
       INTEGER2: XRBufferLength = XRServiceNumber % overlay for append routines
       INTEGER2: XRLengthMsg           % total length of following parameters
ENDRECORD
                                       % parameter formats
TYPE XROUTPARAMHDR = RECORD PACK       % base record for parameter variants
       BYTE: XRPCode                   % parameter code
       BYTE: XRPLength                 % parameter length
ENDRECORD
TYPE XROUTSTRING = XROUTPARAMHDR RECORD PACKED % string parameter format
       BYTES: XRPString(0:XMMAXNameLength-1)   % MAXIMUM length shown
ENDRECORD
TYPE XROUTINTEGER = XROUTPARAMHDR RECORD PACKED % integer parameter format
       INTEGER2: XRPInteger
ENDRECORD
TYPE XROUTDOUBLE = XROUTPARAMHDR RECORD PACKED % integer4 parameter format
       INTEGER4: XRPDouble
ENDRECORD
