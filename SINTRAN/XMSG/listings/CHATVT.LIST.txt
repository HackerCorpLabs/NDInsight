     1          % =========================================================================
     2          % CHATVT - CHATSV's OWN send side (buildFromSlot + broadcast), offline.
     3          %
     4          % WHY THIS EXISTS. The repeat-first-message handoff (2026-09-01) names the
     5          % one untested area: the server's send path. CHATCTST proved the client's
     6          % cache, paint and parse; CHATKT proved the raw kernel get/wri/send and
     7          % rcv/rea/rel sequences deliver distinct content faithfully, even with a
     8          % received message held unreleased. What has never been RUN with synthetic
     9          % distinct inputs is CHATSV's own buildFromSlot and broadcast.
    10          %
    11          % THE METHOD. Link against the real CHATSV.BRF, plant three fake members
    12          % in its member table whose magic numbers are OUR OWN test port, then:
    13          %
    14          %   part A  call buildFromSlot three times with three different slots and
    15          %           texts and DECODE outBuf after each call with CHATLIB's own
    16          %           cm* readers - the build half, no XMSG involved;
    17          %   part B  call broadcast and count + decode what actually arrives on
    18          %           the port - audience selection AND delivered bytes;
    19          %   part C  three build+broadcast rounds with distinct texts, checking
    20          %           after each round that what arrived is THAT round's text -
    21          %           the exact shape of the live bug.
    22          %
    23          % Every message received is dumped (kind, name length, first name byte,
    24          % text length, first text byte) whether it passes or not, so a failing
    25          % transcript identifies the wrong byte without another run.
    26          %
    27          % HOW TO READ IT: every line starts PASS or FAIL, the last line is a
    28          % count. A FAIL in part A is buildFromSlot; a wrong COUNT in part B is
    29          % broadcast's audience filter; wrong CONTENT in part C with A and B clean
    30          % is the live bug caught between build and delivery.
    31          % =========================================================================

   839   (179)      % Receive ONE message into tstIn the way the client does. Answers 0 on
   840   (180)      % success, the receive status otherwise (16896 XMXENTM = port empty).
   841   (181)      ROUTINE VOID, INTEGER : recvOne
   842   (182)          INTEGER : i, ok
   843   (183)          XMMSGIDENTIFIER : ident
   844   (184)          FOR i IN 0:255 DO
   845   (185)              zByte(0) =: tstIn(i)
   846   (186)          ENDFOR
   847   (187)          xmpfrcv(0, myPort, msgType, remotePortHash, ident, nBytes) =: rcvSt
   848   (188)          % XMOK *or* a message type 1..6 - both are a successful receive.
   849   (189)          0 =: ok
   850   (190)          IF rcvSt = XMOK THEN
   851   (191)              1 =: ok
   852   (192)          ENDIF
   853   (193)          IF rcvSt >= 1 AND rcvSt <= 6 THEN
   854   (194)              1 =: ok
   855   (195)          ENDIF
   856   (196)          IF ok = 0 THEN
   857   (197)              rcvSt RETURN
   858   (198)          ENDIF
   859   (199)          nBytes =: readAmount
   860   (200)          IF readAmount > 256 THEN
   861   (201)              256 =: readAmount
   862   (202)          ENDIF
   863   (203)          xmpfrea(0, 0, &
   864   (204)              ADDR(tstIn(0)) FORCE XMUSERADDRESS, &
   865   (205)              0, readAmount, readLength) =: reaSt
   866   (206)          xmpfrel(0, ident) =: relSt
   867   (207)          0 RETURN
