     1          % =========================================================================
     2          % CHATTEST - checks CHATLIB's codec, and prints PASS or FAIL for each case.
     3          %
     4          % THIS IS WHAT THE LIBRARY IS FOR. The client is 6400 lines and takes about
     5          % twenty minutes to compile, so until now every change to the wire format was
     6          % tested by building the whole product and driving two live machines. This
     7          % program links CHATLIB, builds in seconds, needs no XMSG, no server and
     8          % nobody typing - and it fails LOUDLY, on screen, with the numbers.
     9          %
    10          % HOW TO READ IT: every line starts PASS or FAIL, and the last line is a
    11          % count. A single FAIL means the codec is wrong; do not go near a build of the
    12          % real product until this is clean.
    13          %
    14          % ADDING A CASE is one call to `check`. Keep them small and keep the expected
    15          % value a LITERAL - a test that computes its expectation the same way the code
    16          % does will agree with a bug.

   105              % One check. Prints the name, then PASS, or FAIL with both numbers - a
   106              % failure that does not say what it got is a failure you have to reproduce
   107              % by hand.
   108              ROUTINE VOID, VOID (BYTES, INTEGER, INTEGER) : check(what, got, want)
   109                  IF got = want THEN
   110                      passed + 1 =: passed
   111                      OUTPUT(1, 'AL', 'PASS ')
   112                      OUTPUT(1, 'AL', what)
   113                      OUTPUT(1, 'AL', '$')
   114                      RETURN
   115                  ENDIF
   116          
   117                  failed + 1 =: failed
   118                  OUTPUT(1, 'AL', 'FAIL ')
   119                  OUTPUT(1, 'AL', what)
   120                  OUTPUT(1, 'AL', '  got ')
   121                  OUTPUT(1, 'I', got)
   122                  OUTPUT(1, 'AL', '  want ')
   123                  OUTPUT(1, 'I', want)
   124                  OUTPUT(1, 'AL', '$')
   125              ENDROUTINE

   127              % The same check for a BOOLEAN answer. PLANC will not pass a BOOLEAN where
   128              % an INTEGER is declared, so cmSane needs its own - and printing "got 1
   129              % want 0" would be less use than TRUE and FALSE anyway.
   130              ROUTINE VOID, VOID (BYTES, BOOLEAN, BOOLEAN) : checkB(what, got, want)
   131                  IF got = want THEN
   132                      passed + 1 =: passed
   133                      OUTPUT(1, 'AL', 'PASS ')
   134                      OUTPUT(1, 'AL', what)
   135                      OUTPUT(1, 'AL', '$')
   136                      RETURN
   137                  ENDIF

   139                  failed + 1 =: failed
   140                  OUTPUT(1, 'AL', 'FAIL ')
   141                  OUTPUT(1, 'AL', what)
   142                  IF got THEN
   143                      OUTPUT(1, 'AL', '  got TRUE want FALSE')
   144                  ELSE
   145                      OUTPUT(1, 'AL', '  got FALSE want TRUE')
   146                  ENDIF
   147                  OUTPUT(1, 'AL', '$')

   150              % ---- CAN A ROUTINE ASK ITS ARRAY PARAMETER HOW BIG IT IS? --------------
   151              %
   152              % THIS DECIDES THE SHAPE OF THE WHOLE LIBRARY, so it is asked of the machine
   153              % rather than assumed.
   154              %
   155              % cmEnc is told a buffer size by its caller and has to believe it. A caller
   156              % that passes the wrong number gets a silent overflow, because PLANC checks
   157              % no array bound - and one of the tests in this very file did exactly that,
   158              % claiming 512 for a 256-byte buffer.
   159              %
   160              % Three BYTES declared FROM A LITERAL, which is a different shape from an
   161              % array with explicit bounds. 'HELP' is 4, 'DROP-MEMBER' is 11 and
   162              % 'RESTART-TRUNK' is 13 - all three are real CHAT-MON keywords, so if the
   163              % bounds of this shape ever change these tests fail before a menu does.
   164              BYTES : litShort := 'HELP'
   165              BYTES : litVerb  := 'DROP-MEMBER'
   166              BYTES : litLong  := 'RESTART-TRUNK'

   168              % If MAXINDEX works on a PARAMETER, the library can ask each array its own
   169              % size and stop trusting the caller at all. ND-60.117.5 3.17 says MAXINDEX
   170              % returns the DECLARED UPPER BOUND, and page 249 says it is NOT available on
   171              % array parameters in a STANDARD routine - which implies it IS in an ordinary
   172              % one. "Implies" is not a measurement, so these three checks measure it.
   173              ROUTINE VOID, INTEGER (BYTES) : askSize(a)
   174                  MAXINDEX(a, 1) RETURN

   354                  % ---- THE BOUNDED READERS -----------------------------------------

   356                  % cmLenN and cmSane are what a receive loop should use: they are told

   358                  % here is one that the unbounded readers answer WRONGLY but plausibly.

   388                  % ---- THE NAME MUST BE CLAMPED TOO --------------------------------
   389                  %
   390                  % cmEnc clamped the TEXT against bufMax and never the NAME. With a
   391                  % 255-byte name and a 256-byte buffer the text clamped to nothing, and
   392                  % then the name loop wrote buf(2)..buf(256) and the two length bytes
   393                  % landed at buf(257) and buf(258) - THREE BYTES PAST THE END, silently.
   394                  %
   395                  % Not reachable from the wire today: the client sends its own nickname
   396                  % and the server drops a join whose name is longer than 16. But that
   397                  % safety lives in the CALLERS, which is the exact duplication this
   398                  % library exists to remove - so it is enforced here and tested here.
   399                  %
   400                  % A message can never be longer than the buffer it was written into.
   401                  % That single check is the one that would have caught the overflow.
   402                  % The clamp has ONE right answer, so it is checked exactly rather than
   403                  % as "small enough": a name must leave room for the 2-byte header and
   404                  % the 2-byte length, so the most that fits a 256-byte buffer is 252,

   412                  check('huge name text at 256   ', cmTxAt(buf), 256)

   417                  check('cmEncAt total is 256    ', len, 256)

   427                  %
   428                  % If all three pass, the library can bound itself and no caller can lie
   429                  % to it about a size. If the subarray case fails, it can still bound
   430                  % itself for whole arrays, which is how every current caller passes them.
   431                  check('MAXINDEX on a parameter ', askSize(buf), 255)
   432                  check('MAXINDEX sees a SMALLER ', askSize(nm), 15)
   433                  check('MAXINDEX on a SUBARRAY  ', askSize(tx(0:9)), 9)
   434          
   435                  % ---- AND ON A BYTES DECLARED FROM A LITERAL? ----------------------
   436                  %
   437                  % THIS IS THE CASE THE WHOLE PRODUCT NOW LEANS ON. On 2026-08-31 the
   438                  % length parameter was removed from six helpers - putWord, logLine,
   439                  % buildAdmText, cmdIs, tryCmd and showIfMatch - and 93 hand-typed
   440                  % numbers went with it. Each now does MAXINDEX(text, 1) + 1 instead.
   441                  %
   442                  % Two of those, tryCmd and showIfMatch in CHAT-MON, are handed a
   443                  % MODULE-LEVEL BYTES declared from a literal rather than a literal at
   444                  % the call site. R115 measured MAXINDEX on array parameters and on a
   445                  % subarray; it did NOT measure this. The bounds SHOULD be 0:len-1, so
   446                  % MAXINDEX + 1 should be the length - but "should" is not measured, and
   447                  % if it is wrong the symptom is a menu where a command quietly stops
   448                  % matching, which reads as a logic bug and not as an array question.
   449                  %
   450                  % Three lengths on purpose: a short word, one that is not a round
   451                  % number, and one long enough to cross a byte boundary in a way a
   452                  % single case could not distinguish.
   453                  check('MAXINDEX on a literal   ', askSize(litShort) + 1, 4)
   454                  check('MAXINDEX literal, longer', askSize(litVerb) + 1, 11)
   455                  check('MAXINDEX literal, 13 chr', askSize(litLong) + 1, 13)
   456          
   457                  % ---- THE DECODERS AGAINST THE ARITHMETIC THEY REPLACE -------------
   458                  %
   459                  % WHY THIS GROUP EXISTS. The client and the server each parse the header
   460                  % by hand, in three places between them:
   461                  %
   462                  %       2 + nameLength =: textAt
   463                  %       inBuf(textAt) * 256 + inBuf(textAt + 1) =: textLength
   464                  %       textAt + 2 =: textAt
   465                  %
   466                  % Replacing that with cmTxAt and cmTxLen is only safe if the library
   467                  % computes the SAME numbers for every name length, not just for the one
   468                  % five-character nickname the tests above happen to use. A name length is
   469                  % a byte off a wire and can be anything from 0 to 255, so the two forms
   470                  % are compared across the whole range here rather than at one point.
   471                  %
   472                  % A ZERO-LENGTH NAME IS IN THE RANGE ON PURPOSE. It is the shortest legal
   473                  % header and the one an off-by-one lands on first.
   474                  %
   475                  % THE UPPER BOUND IS THE BUFFER, NOT 255. A header with name length n puts
   476                  % its two length bytes at n+2 and n+3, so n = 255 would write THREE BYTES
   477                  % PAST THE END of buf(0:255) - and PLANC checks no array bounds, so it
   478                  % would corrupt whatever sits after it and still print PASS. Asked with
   479                  % MAXINDEX rather than written as a literal, so resizing buf cannot leave
   480                  % a stale number here. (MAXINDEX gives the DECLARED UPPER BOUND, 255.)

   489                      % filling 255 of them for each of 256 rounds is work for nothing.

   492                      44 =: buf(2 + i + 1)         % text length LOW byte - so 300, over 255

   495                      IF cmTxAt(buf) >< 2 + i + 2 THEN
   496                          check('cmTxAt matches by hand  ', cmTxAt(buf), 2 + i + 2)
   497                      ENDIF

   499                      % cmTxLen must equal the hand-rolled high * 256 + low. 300 is chosen
   500                      % because a decoder that read ONLY the low byte would answer 44 and
   501                      % pass every other test in this file - no message the product has
   502                      % ever sent reached 256 characters.

   504                          check('cmTxLen matches by hand ', cmTxLen(buf), 300)
   505                      ENDIF

   511                  % the COUNT it covered, so a bound that silently shrinks is visible.

   735                  %
   736                  % THIS IS THE ONE ROUTINE HERE THAT STOPS A WRITE OVERFLOW rather than
   737                  % a wrong answer, so it is tested against FOUR DIFFERENT BUFFERS. That
   738                  % is the whole point: if MAXINDEX did not really ask each array, every
   739                  % one of these would answer the same and the differing sizes below
   740                  % would fail. It is the same proof as the array-parameter probe above,
   741                  % applied to the routine that matters most.
   742                  check('clamp 10 into nm(16)    ', xrClamp(10, nm), 10)
   743                  check('clamp 99 into nm(16)    ', xrClamp(99, nm), 16)
   744                  check('clamp 99 into tx(64)    ', xrClamp(99, tx), 64)
   745                  check('clamp 99 into buf(256)  ', xrClamp(99, buf), 99)
   746                  check('clamp 999 into buf(256) ', xrClamp(999, buf), 256)
   747                  check('clamp 999 into bigBuf   ', xrClamp(999, bigBuf), 512)

   751                  % the off-by-one that a hand-written "IF want > inBufSize" gets right
   752                  % only because the constant happens to say 256 and not 255.
   753                  check('clamp 256 into buf(256) ', xrClamp(256, buf), 256)
   754                  check('clamp 257 into buf(256) ', xrClamp(257, buf), 256)

   761                  check('clamp -32000 into buf   ', xrClamp(-32000, buf), 0)

   763                  % ---- ADDR OF A PARAMETER *IS* THE CALLER'S ARRAY. ANSWERED. ---------
   764                  %
   765                  % ASKED because the XMSG transport was about to move into a shared
   766                  % library, and a shared routine has to take ADDR of a PARAMETER - which
   767                  % nothing in this product had ever done. Every other ADDR here is taken
   768                  % of a MODULE-LEVEL array, in the same module as the call.
   769                  %
   770                  % THE ANSWER IS YES, measured on D100 2026-08-27:
   771                  %
   772                  %     ADDR param  1001        ADDR here  1001
   773                  %
   774                  % Identical, for all three arrays. So a routine may take the address of
   775                  % a buffer it was handed and give it to XMPFREA or XMPFWRI.
   776                  %
   777                  % IT TOOK THREE GOES, AND THE TWO WRONG ONES ARE THE LESSON.
   778                  %
   779                  % Attempt 1 forced both addresses to a bare INTEGER4 and got
   780                  % 3473663 against 65791 - different in both halves. They reproduced
   781                  % EXACTLY across two separate builds, which read as reliability, and
   782                  % the conclusion written down was "a parameter address is not the
   783                  % caller's array, the transport cannot be shared". Wrong.
   784                  %
   785                  % Attempt 2 assumed the trouble was the bare type and switched to a
   786                  % NAMED one, TYPE xrAddr = INTEGER4. Same two numbers, same warning.
   787                  % Also wrong, and wrong for the same unexamined reason.
   788                  %
   789                  % What both had was a WIDTH mismatch, and the pulled listing said so
   790                  % every time - a WARNING, not an error, on the call site as well as
   791                  % inside the routine:
   792                  %
   793                  %     715/CTMAIN  *** WARNING - ILLEGAL DATA-ELEMENT TO BE CONVERTED
   794                  %
   795                  % AN ADDRESS ON THIS MACHINE IS SIXTEEN BITS. XMP-B02:IMPT declares
   796                  %
   797                  %     TYPE XMUSERADDRESS = INTEGER          % NB: NEW MC68000 compiler!!
   798                  %
   799                  % and the line ABOVE it, which is where the mistake came from, is a
   800                  % COMMENT reading "modify XMUSERADDRESS = INTEGER4" - an instruction to
   801                  % widen it FOR THE MC68000. Taking that commented instruction for the
   802                  % declaration is what produced a 32-bit force. The extra word was
   803                  % adjacent memory, which is why the junk was stable enough to look like
   804                  % a measurement.
   805                  %
   806                  % THREE THINGS WORTH KEEPING:
   807                  %
   808                  %   a WARNING can invalidate a result as thoroughly as an ERROR - which
   809                  %   is why the gate prints warnings and not only errors;
   810                  %
   811                  %   reproducible is not the same as correct;
   812                  %
   813                  %   read the DECLARATION, not the first line that matches the name. A
   814                  %   commented-out instruction sat one line above the real one.
   815                  xrAddrOf(buf) =: addrA
   816                  ADDR(buf(0)) FORCE xrAddr =: addrB
   817          
   818                  % PRINT THEM BOTH BEFORE JUDGING. The check below failed on its first
   819                  % run, and there are two quite different reasons it could:
   820                  %
   821                  %   the parameter address really is not the caller's array - in which
   822                  %   case the transport cannot be shared in this shape at all; or
   823                  %
   824                  %   ADDR gives a POINTER and FORCE INTEGER4 reads a word of whatever
   825                  %   sits next to it, so the comparison is measuring garbage in the high
   826                  %   half while the address itself is perfectly fine.
   827                  %
   828                  % The working product forces ADDR to XMUSERADDRESS, which XMP-B02:IMPT
   829                  % declares as INTEGER4, so the idiom itself is sound. Printing the
   830                  % numbers says which of the two this is; reasoning about it does not.
   831                  OUTPUT(1, 'AL', 'ADDR param  ')
   832                  OUTPUT(1, 'I', addrA)
   833                  OUTPUT(1, 'AL', '   ADDR here  ')
   834                  OUTPUT(1, 'I', addrB)
   835                  OUTPUT(1, 'AL', '$')
   836          
   837                  checkB('param addr = here buf   ', addrA = addrB, TRUE)
   838          
   839                  xrAddrOf(nm) =: addrA
   840                  ADDR(nm(0)) FORCE xrAddr =: addrB
   841                  checkB('param addr = here nm    ', addrA = addrB, TRUE)
   842          
   843                  xrAddrOf(bigBuf) =: addrA
   844                  ADDR(bigBuf(0)) FORCE xrAddr =: addrB
   845                  checkB('param addr = here bigBuf', addrA = addrB, TRUE)
   846          
   847                  % Three separate arrays cannot share a start address, so these catch a
   848                  % routine that ignores its parameter and hands back one fixed thing.
   849                  % They PASSED even under the illegal conversion, which is exactly why
   850                  % they were never evidence that the rest of the probe was sound.
   851                  checkB('buf and nm differ       ', xrAddrOf(buf) = xrAddrOf(nm), FALSE)
   852                  checkB('nm and bigBuf differ    ', xrAddrOf(nm) = xrAddrOf(bigBuf), FALSE)
   853          
   854                  % ---- the count ---------------------------------------------------
   855                  OUTPUT(1, 'AL', '$')
   856                  OUTPUT(1, 'AL', 'passed ')
   857                  OUTPUT(1, 'I', passed)
   858                  OUTPUT(1, 'AL', '   failed ')
   859                  OUTPUT(1, 'I', failed)
   860                  OUTPUT(1, 'AL', '$')
   861          
   862                  IF failed > 0 THEN
   863                      OUTPUT(1, 'AL', '*** THE CODEC IS WRONG - do not build the product$')
   864                  ELSE
   865                      OUTPUT(1, 'AL', 'codec is good$')
   866                  ENDIF
